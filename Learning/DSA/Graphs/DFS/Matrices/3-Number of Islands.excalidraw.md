function numIslands(grid):
    directions = [[-1,0], [1,0], [0,-1], [0,1]]
    stack = []
    visited = new Set()
    count = 0
    function dfsIterative(row, col):
        stack.push([row,col])
        while stack.length != 0:
            [ir, ic] = stack.pop()
            key = `{ir}-{ic}`
            if visited.has(key):
                continue
            if ir < 0 or ir >= grid.length or ic < 0 or ic >= grid[0].length:
                continue
            if grid[ir][ic] == 0:
                continue
            visited.add(key)
            for [dr, dc] of directions:
                key = `{ir+dr}-{ic+dc}`
                if !visited.has(key):
                    stack.push([ir+dr, ic+dc])
    for row = 0 to grid.length - 1:
        for col = 0 to grid[0].length - 1:
