function validTree(n, edges):
    if edges.length != n - 1:
        return false
    adjList = {}
    for i = 0 to n - 1:
        adjList[i] = []
    
    for i = 0 to edges.length - 1:
        node1 = edges[i][0]
        node2 = edges[i][1]
        adjList[node1].append(node2)
        adjList[node2].append(node1)
    visited = {}
    function dfs(node):
        visited[node] = true
        
        for neighbour in adjList[node]:
            if neighbour not in visited:
                dfs(neighbour)

        return false

function validTree(n, edges):
    if edges.length != n - 1:
        return false
    adjList = {}
    for i = 0 to n - 1:
        adjList[i] = []
    
    for i = 0 to edges.length - 1:
        node1 = edges[i][0]
        node2 = edges[i][1]
        adjList[node1].append(node2)
        adjList[node2].append(node1)
    visited = {}
    function dfsIterative(start):
        visited[start] = true
        stack = []
        stack.push(start)
        while stack.length != 0:
            node = stack.pop()
            for neighbour in adjList[node]:
                if neighbour not in visited:
                    visited[neighbour] = true                    
                    stack.push(neighbour)
                
    dfs(0)
    if len(visited.keys()) != n:
        return false
    return true          ^8T4z0OxW
