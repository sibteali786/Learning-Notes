def insertionSort(nums: Array[int]) return Array[int] {
    for i = 0 to nums.length - 1 {
        for j = i + 1 down to 1 {
            if nums[j] < nums[j-1]  {
                swap(nums, j, j-1)
            }else {
                break
            }
        }
    }
    return nums
}

def swap(nums: Array[int], first: int, second: int) {
    temp = nums[first]
    nums[first] = nums[second]
    nums[second] = temp
} ^vGLrtp1F
