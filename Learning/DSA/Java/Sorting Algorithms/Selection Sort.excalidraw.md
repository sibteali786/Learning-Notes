def selectionSort(nums: Array[int]) returns sorted Array[int] {
    for i = 0 to nums.length - 1 {
        lastIndex = nums.length - i - 1
        maxNumberIndex = getMaxNumberIndex(nums, 0, lastIndex)
        swap(nums, maxNumberIndex, lastIndex)
    }
}

def getMaxNumberIndex(nums: Array[int], start, end){
    max = start
    for i = start + 1 to less than equal to end {
        if nums[max] < nums[i] {
            max = i
        }
    }    
    return max
}

def swap(nums: Array[int], first: int, second: int) {
    temp = nums[first]
    nums[first] = nums[second]
    nums[second] = temp
