function maxScore(nums, k) {
    minSum = 0
    if ( nums.length != k ) {
        minSum = Infinity
        minSubArraySize = nums.length - k
        for i = 0 to nums.length - minSubArraySize {
            sum = nums[i]
            for j = i + 1 to i + minSubArraySize - 1 {
                sum = sum + nums[j]
            }
            minSum = Math.min(minSum, sum)
        }
    }
    // Get total sum
    totalSum = 0
    for i = 0 to nums.length - 1 {
        totalSum = totalSum + nums[i]
    }

    return totalSum - minSum
} ^ZROuVz47

function maxScore(cards, k) {
    const total = cards.reduce((sum, card) => sum + card, 0);
    if (k === cards.length) {
        return total;
    }
    let state = 0;
    let maxPoints = 0;
    let start = 0;
    for (let end = 0; end < cards.length; end++) {
        state += cards[end];
        if (end - start + 1 === cards.length - k) {
            maxPoints = Math.max(total - state, maxPoints);
            state -= cards[start];
            start++;
        }
    }
    return maxPoints;
} ^FH11buBu
