function trap(nums){
    totalWaterStored = 0
    
    for i = 0 to nums.length - 1 {
        maxLeft = 0
        maxRight = 0
        // find max to left of current index
        if ( i != 0 ) {
            for j = 0 to i {
                maxLeft = max(nums[j], maxLeft)
            }
        }
        if ( i != nums.length - 1){
            for j = nums.length - 1 down to i {
