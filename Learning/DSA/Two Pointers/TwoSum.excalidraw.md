function twoSum(numbers, target) {
    let left = 0, right = numbers.length - 1
    let sum = 0
    while (left < right) {
        sum = numbers[left] + numbers[right]
        if ( sum == target) {
            return [left+1, right+1]    
        } else if ( sum < target ) {
            left = left + 1
        }else {
            right = right - 1
        }
    }
}  ^3FNZjiJh

Time Complexity is O(N) as loop goes over each element at once in worst case
Space complexity is O(1) since each variable takes constant auxillary space. ^5lSqX9Hc
