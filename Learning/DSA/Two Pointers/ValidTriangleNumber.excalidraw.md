function triangleNumber(nums: Array[integer]) returns integer {
    // sort array to optimise fidning triplets
    nums.sort((a,b) => a-b)
    count = 0
    for i = nums.length - 1 down to 2 {
        left = 0
        right = i - 1
        while ( left < right ) {
            sum = nums[left] + nums[right]
            if ( sum > nums[i] ) {
                count += right - left // becuase all values from left till right would be greater than i since left is smallest in sorted 
                right = right - 1
            }else {
                left = left + 1
            } 
        }
    }
    return count
}

/**

var triangleNumber = function(nums) {
    nums.sort((a,b) => a-b);
    let count = 0;
    for (let i = nums.length - 1; i >= 2; i--) {
        let left = 0, right = i - 1;
        while (left < right) {
            let sum = nums[left] + nums[right]
            if (sum > nums[i]) {
                count += right - left
                right--
            }else{
                left++
            }
        }
    }
    return count;
}; ^kM6SDI67

- Sorting O9(nlogn)
- Since one outer loop at worst O(N-2) times so O(N)
- one inner loop which runs O(N-i) where i is current index which alomsot reaches to N
so overall O(N) * O(N) + O(nlogN) = O(N^2)
