def hasDuplicate(nums: List[int]) returns boolean{
    dictionary = {}

    }

class Solution:
    def hasDuplicate(self, nums: List[int]) -> bool:
        myDict = {}
        for num in nums:
            if num in myDict.keys():
                return True
            else:
