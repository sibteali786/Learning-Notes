function firstMissingPositive(nums):
    placeholder = nums.length + 1
    for i = 0 to nums.length - 1:
        if nums[i] <= 0 OR nums[i] >= placeholder:
            nums[i] = placeholder // mark as irrelevant
    
    // mark as those who exist in the array
    for i = 0 to nums.length - 1:
        if Math.abs(nums[i]) < placeholder:
            v = abs(nums[i]) - 1
            if nums[v] < 0: continue
            nums[v] = -1 * nums[v]
    
    // find first non negative
    for i = 0 to nums.length - 1:
        if nums[i] > 0:
            return i + 1
    return placeholder ^hCIREDle
