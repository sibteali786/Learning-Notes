def isAnagram(s: string, t: string) returns boolean {
    dictA = {} 
    for char of s {
        if char in dictA {
            dictA[char] += 1    
        }else {
            dictA[char] = 1
        }
    }
    
    for char of t {
        if char in dictA.keys() {
            dictA[char] -= 1
        }else {
            return False
        }
    }

    for value in dictA.values() {
        if value != 0 {
            return False
        }
    }

    return True
} ^cYXz5qwa
