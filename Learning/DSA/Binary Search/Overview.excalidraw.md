function binarySearch(nums, target) {
    let left = 0
    let right = nums.length - 1
    while ( left <= right ) {
        let mid = left + ( right - left ) / 2
        let midElement = nums[mid]
        if ( midElement === target ) {
            return mid
        }
        
        if ( midElement < target ){
            left = mid + 1
        }else {
            right = mid - 1
        }         
    }
    
    return -1
} ^Df9NYTqt
