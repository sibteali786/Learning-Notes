function bianrySearch(nums, target) {
    left = 0
    right = nums.length - 1
    while (left <= right) {
        mid = Math.floor((left+right)/2);
        if (nums[mid] == target){
            return mid;
        }
        if(nums[mid] > target){
            right = mid - 1
        }else {
            left = mid + 1
        }
    }

    return -1
