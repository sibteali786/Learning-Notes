var kthSmallest = function(matrix, k) {
    let n = matrix.length
    let left = matrix[0][0]
    let right = matrix[n-1][n-1]
    
    while( left <= right ) {
        let mid = Math.floor(left + ((right - left )/2))
        count = countLessOrEqual(matrix, mid)
        
        if (count < k ) {
            left = mid + 1
        }else {
            right = mid - 1
        }
    }
    return left
};

function countLessOrEqual(matrix, mid) {
    let n = matrix.length
    let count = 0
    let row = n - 1
    let col  = 0
    while ( row >= 0 && col < n ) {
        if ( matrix[row][col] <= mid ) {
            count += row + 1
            col++
        }else {
            row--
        }
    }
    
    return count
}  ^sZR9CLyI

Time Complexity: O(n * log(max - min)) where n is the matrix dimension and max - min is the range of values. Binary search runs O(log(max - min)) iterations, and each iteration does O(n) work for the staircase counting.

Space Complexity: O(1) We only use a few variables (left, right, mid, row, col, count) regardless of matrix size. ^FSm5EYgF
