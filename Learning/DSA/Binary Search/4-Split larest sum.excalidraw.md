function(nums, k): 
    // find max of array -> start and sum of all -> end
    left = Math.max(...nums)
    right = nums.reduce((acc, curr) => acc + curr, 0)
    
    while ( left <= right ): 
        mid = floor of ( left + ( (right - left ) / 2) ) 
        subArrays = getSubArrays(nums, mid)
        if subArrays <= k:
            right = mid - 1
        else:
