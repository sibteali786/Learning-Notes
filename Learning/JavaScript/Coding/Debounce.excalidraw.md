Implementing Debounce in Javascript ^D4l7vfZz

function debounce(fn, wait, options){    
    let leadingTimeoutId = null
    let trailingTimeoutId = null
    const {trailing = true, leading = false} = options ?? {};
    return function (...args) {
        const context = this // capture this at call time 
        if ( leading === true) {
            // check if first call
            if ( leadingTimeoutId === null ) {
                fn.apply(context, args)    
            }else {
                clearTimeout(leadingTimeoutId)
            }       
            leadingTimeoutId = setTimeout(() => {
                leadingTimeoutId = null
            }, wait)
        }
        if (trailing === true){
            clearTimeout(trailingTimeoutId) // if there is no timeout it fails silently
            trailingTimeoutId = setTimeout(() => fn.apply(context, args), wait)
        }
        
    }
} ^TnQ41cZv

/**
 * @param {Function} func
 * @param {number} wait
 * @return {Function}
 */
export default function debounce(func, wait) {
  let timeoutId = null;
  let storedContext = undefined;
  let storedArgs = undefined;
  const myFunction = function (...args) {
    clearTimer();
    storedContext = this;
    storedArgs = args;
    // set new timer
    timeoutId = setTimeout(function () {
      invoke();
    }, wait);
  };
  function clearTimer() {
    clearTimeout(timeoutId);
    // cleanup
    timeoutId = null;
  }

  function invoke() {
    if (timeoutId == null) {
      return;
    }
    clearTimer();
    func.apply(storedContext, storedArgs);
  }
  myFunction.cancel = clearTimer;
  myFunction.flush = invoke;
  return myFunction;
}
